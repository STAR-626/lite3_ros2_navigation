cmake_minimum_required(VERSION 3.8)
project(lite3_perception)
find_package(ament_cmake REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(pluginlib REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
add_library(lite3_flat_ground_voxel_layer SHARED src/flat_ground_voxel_layer.cpp src/forecast_layer.cpp src/route_penalty_layer.cpp)
target_compile_features(lite3_flat_ground_voxel_layer PRIVATE cxx_std_17)
ament_target_dependencies(lite3_flat_ground_voxel_layer
  nav2_costmap_2d pluginlib sensor_msgs nav_msgs rclcpp tf2 tf2_ros)
pluginlib_export_plugin_description_file(nav2_costmap_2d plugins.xml)
install(TARGETS lite3_flat_ground_voxel_layer LIBRARY DESTINATION lib)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
install(FILES LICENSE DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/pointcloud_obstacles.py DESTINATION lib/${PROJECT_NAME}
  RENAME pointcloud_obstacles)
install(FILES scripts/motion_forecast.py DESTINATION lib/${PROJECT_NAME})
ament_package()
